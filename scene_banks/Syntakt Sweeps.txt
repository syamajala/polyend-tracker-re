# OpenTracker scene bank
# Plays an Elektron Syntakt track over MIDI next to the samples. Set MIDI
# instrument 1 to the Syntakt track's channel and load the Syntakt MIDI
# preset on it (slots A-F: Cutoff, Reso, Filter env, Amp decay, Delay send,
# Reverb send). Columns 1-6 send those CCs; 7-12 act on the sample tracks
# and the mix. A = 1 (Home), B = 2 (Open). Every scene sets all six CC
# columns, because a CC has no 'off' to fade to.

# Columns: type: row 1, row 2, row 3
# 1 Cutoff: closed, half, open
# 2 Reso
# 3 Filter env: none, some, lots
# 4 Amp decay: short, mid, long
# 5 Delay send
# 6 Reverb send
# 7 the sample tracks
# 8 the sample tracks
# 9 rolls (samples and MIDI): 1 step, 1/4, 1/8
# 10 loops 4, 2 or 1 step
# 11 the Tracker's whole mix (not the Syntakt)
# 12 the Tracker's mix: 1 beat, 1 bar, 2 bars
column 1 = MIDI CC: 20 64 115 @ midi 1 slot A
column 2 = MIDI CC: 0 50 100 @ midi 1 slot B
column 3 = MIDI CC: 64 90 120 @ midi 1 slot C
column 4 = MIDI CC: 20 60 110 @ midi 1 slot D
column 5 = MIDI CC: 0 50 100 @ midi 1 slot E
column 6 = MIDI CC: 0 50 100 @ midi 1 slot F
column 7 = Sample LP cutoff: -25 -50 -75
column 8 = Sample reverb snd: 25 50 90
column 9 = Step Repeater: 8 11 13
column 10 = Pattern Length: 3 2 1
column 11 = DJ filter: -50 40 75
column 12 = Tape stop: 3 5 6

# Scenes: a row 0..3 per column (0 = the scene leaves it alone)
# 1 Home: a plain, mid-open sound (A)
scene 1 = 2 1 1 2 1 1 0 0 0 0 0 0
# 2 Open: filter open, longer, wetter (B)
scene 2 = 3 2 2 3 2 2 0 0 0 0 0 0
# 3 Pluck: closed filter, big envelope, short
scene 3 = 1 2 3 1 2 1 0 0 0 0 0 0
# 4 Acid: squelchy: closed, resonant
scene 4 = 1 3 3 2 1 1 0 0 0 0 0 0
# 5 Pad: long and washed out
scene 5 = 2 1 1 3 3 3 0 0 0 0 0 0
# 6 Breakdown: everything dark and roomy
scene 6 = 1 1 1 3 2 3 2 2 0 0 0 0
# 7 Build: open and resonant, rolling, mix thinned
scene 7 = 3 3 2 1 3 2 0 0 3 0 2 0
# 8 Stutter: one step on repeat
scene 8 = 2 2 2 1 1 1 0 0 2 3 0 0

a = 1
b = 2
