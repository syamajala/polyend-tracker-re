# Song repeat counts (#10): runs on the simulator (opentracker-sim --serve)
# and on a Tracker, from a fresh project with remote control on.
#   tools/tracker_remote.py PORT tools/remote_scripts/song_count.txt
# Song: pattern 1 x1, then pattern 2 x3. Waits follow the sequencer
# (wait-for), so the tempo and pattern length do not matter.
tap SONG
wait 300
expect-screen Song
tap REC              # edit mode
tap F4               # slot 2
tap F7               # slot 2: pattern 2
tap RIGHT            # Count column
enc 2                # x3
wait 200
expect-text Count
expect-text x3
tap LEFT
tap REC              # edit off
tap UP               # slot 1
tap F2               # Play Song
wait-for song == 1 2000
expect slot == 1
# Slot 2 plays three times, the pass shown in the Count column.
wait-for slot == 2 20000
expect pass == 1
expect-text 1/3
wait-for pass == 2 20000
expect slot == 2
expect-text 2/3
wait-for pass == 3 20000
expect-text 3/3
wait-for slot == 1 20000   # then the song loops
expect pass == 1
tap F2                     # stop
wait-for play == 0 2000
